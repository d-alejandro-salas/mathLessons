'use client';

import { useState } from 'react';
import Section from './Section';
import { PRECIOS, PROMO_FINDE, fmt } from '@/lib/data';

const { duraciones, filas, destacada } = PRECIOS;
const pct = Math.round(PROMO_FINDE.descuento * 100);
const hl = (d) => (d === destacada ? 'bg-indigo-100' : '');
const cell = 'border-b border-line px-[0.6rem] py-[0.9rem]';

export default function Pricing() {
  const [col, setCol] = useState(duraciones.indexOf(destacada));
  const [cant, setCant] = useState(1);
  const [finde, setFinde] = useState(false);

  // Valores derivados del estado: se calculan en cada render, no hace falta useEffect.
  const conPromo = PROMO_FINDE.activa && finde;
  let total = filas.slice(0, cant).reduce((s, f) => s + f.precios[col], 0);
  if (conPromo) total = Math.round((total * (1 - PROMO_FINDE.descuento)) / 100) * 100;

  return (
    <Section
      id="precios"
      alt
      kicker="Precios"
      title="Valores claros por clase"
      lead="Cuantas más clases tomás en la semana, menor es el valor de cada una. Elegí la duración que mejor se adapte a vos."
    >
      {PROMO_FINDE.activa && (
        <p className="mt-4 rounded-[14px] border border-acc bg-amber-100 px-4 py-[0.8rem] font-semibold text-amber-900">
          🎉 Promo de fin de semana: {pct}% de descuento en clases de sábado y domingo.
        </p>
      )}

      <div className="mt-8 overflow-x-auto rounded-card bg-white shadow-soft">
        <table className="w-full min-w-[420px] border-collapse text-center">
          <caption className="caption-bottom p-[0.8rem] text-[0.85rem] text-slate-500">
            Valores en pesos argentinos (ARS) por clase, según el orden de la clase dentro de la semana.
          </caption>
          <thead>
            <tr>
              <th scope="col" className={`${cell} bg-ink pl-4 text-left text-[0.95rem] text-white`}>Semanalmente</th>
              {duraciones.map((d) => (
                <th key={d} scope="col" className={`${cell} text-[0.95rem] text-white ${d === destacada ? 'bg-brand' : 'bg-ink'}`}>
                  {d} minutos
                </th>
              ))}
            </tr>
          </thead>
          <tbody>
            {filas.map((f) => (
              <tr key={f.etiqueta} className="[&:last-child>*]:border-b-0">
                <th scope="row" className={`${cell} pl-4 text-left text-ink`}>{f.etiqueta}</th>
                {f.precios.map((p, i) => (
                  <td key={i} className={`${cell} font-semibold tabular-nums ${hl(duraciones[i])}`}>{fmt(p)}</td>
                ))}
              </tr>
            ))}
          </tbody>
        </table>
      </div>

      <div className="mt-[1.6rem] grid gap-4 min-[700px]:grid-cols-[1fr_1fr_1.2fr] min-[700px]:items-end">
        <label className="grid gap-[0.3rem] text-[0.9rem] font-bold text-ink">
          Duración de cada clase
          <select
            value={col}
            onChange={(e) => setCol(Number(e.target.value))}
            className="min-h-12 rounded-xl border-2 border-line bg-white p-[0.8rem]"
          >
            {duraciones.map((d, i) => (
              <option key={d} value={i}>{d} minutos</option>
            ))}
          </select>
        </label>

        <label className="grid gap-[0.3rem] text-[0.9rem] font-bold text-ink">
          Clases por semana
          <select
            value={cant}
            onChange={(e) => setCant(Number(e.target.value))}
            className="min-h-12 rounded-xl border-2 border-line bg-white p-[0.8rem]"
          >
            {filas.map((_, i) => (
              <option key={i} value={i + 1}>{i + 1} {i ? 'clases' : 'clase'} por semana</option>
            ))}
          </select>
        </label>

        <div className="rounded-card bg-ink p-[1.2rem] text-center text-white" aria-live="polite">
          <small className="text-indigo-200">Total semanal</small>
          <strong className="block text-[2.2rem] leading-[1.1] text-acc">{fmt(total)}</strong>
          <small className="text-indigo-200">
            ≈ {fmt(Math.round(total / cant))} por clase{conPromo ? ' · promo fin de semana' : ''}
          </small>
        </div>

        {PROMO_FINDE.activa && (
          <label className="col-span-full flex items-center gap-[0.6rem] font-semibold text-ink">
            <input
              type="checkbox"
              checked={finde}
              onChange={(e) => setFinde(e.target.checked)}
              className="size-[22px] accent-brand"
            />
            <span>Mis clases son en fin de semana (−{pct}%)</span>
          </label>
        )}
      </div>
    </Section>
  );
}

'use client';

import { useCallback, useEffect, useRef, useState } from 'react';
import { createPortal } from 'react-dom';

const CARD = 'space-y-3 rounded-xl border border-slate-200 bg-white p-5 shadow-sm';

function ZoomOverlay({ onClose, children }) {
  const cardRef = useRef(null);
  const closeRef = useRef(null);
  const [show, setShow] = useState(false);

  useEffect(() => {
    const el = cardRef.current;
    // Escala la tarjeta para ocupar ~90% de la pantalla (máx. x3), igual que el original.
    const fit = () => {
      el.style.zoom = 1;
      el.style.zoom = Math.min((innerWidth * 0.9) / el.offsetWidth, (innerHeight * 0.9) / el.offsetHeight, 3);
    };
    // La copia se renderiza con LaTeX crudo: hay que tipografiarla antes de medir.
    const mj = window.MathJax;
    const typeset = mj?.startup?.promise ? mj.startup.promise.then(() => mj.typesetPromise([el])) : Promise.resolve();
    typeset.then(fit, fit);
    fit();

    const raf = requestAnimationFrame(() => setShow(true));
    const onKey = (e) => e.key === 'Escape' && onClose();
    document.addEventListener('keydown', onKey);
    document.body.style.overflow = 'hidden';
    closeRef.current?.focus();

    return () => {
      cancelAnimationFrame(raf);
      document.removeEventListener('keydown', onKey);
      document.body.style.overflow = '';
    };
  }, [onClose]);

  return createPortal(
    <div
      role="dialog"
      aria-modal="true"
      onClick={onClose}
      className={`fixed inset-0 z-[100] flex cursor-zoom-out items-center justify-center overflow-hidden bg-slate-900/80 backdrop-blur-sm transition-opacity duration-300 print:hidden ${show ? 'opacity-100' : 'opacity-0'}`}
    >
      <div
        ref={cardRef}
        onClick={(e) => e.stopPropagation()}
        className={`${CARD} w-[28rem] max-w-none flex-none cursor-default shadow-[0_25px_60px_rgba(0,0,0,0.5)] transition-transform duration-300 ${show ? 'scale-100' : 'scale-[0.94]'}`}
      >
        {children}
      </div>
      <button
        ref={closeRef}
        type="button"
        aria-label="Cerrar"
        onClick={onClose}
        className="absolute top-4 right-4 size-10 cursor-pointer rounded-full bg-white text-[1.1rem] font-bold text-slate-900 shadow-[0_4px_12px_rgba(0,0,0,0.4)] hover:bg-slate-200"
      >
        ✕
      </button>
    </div>,
    document.body,
  );
}

/** Bloque del aside que se amplía en un overlay al hacer clic o con Enter/Espacio. */
export default function ExpandableCard({ label, children }) {
  const [open, setOpen] = useState(false);
  const triggerRef = useRef(null);
  const close = useCallback(() => {
    setOpen(false);
    triggerRef.current?.focus();
  }, []);

  return (
    <>
      <div
        ref={triggerRef}
        role="button"
        tabIndex={0}
        title="Click para ampliar"
        aria-label={`Ampliar: ${label}`}
        onClick={() => setOpen(true)}
        onKeyDown={(e) => {
          if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault();
            setOpen(true);
          }
        }}
        className={`${CARD} cursor-zoom-in transition duration-200 hover:-translate-y-0.5 hover:shadow-[0_8px_24px_rgba(37,99,235,0.18)] focus-visible:outline-[3px] focus-visible:outline-offset-2 focus-visible:outline-blue-600`}
      >
        {children}
      </div>
      {open && <ZoomOverlay onClose={close}>{children}</ZoomOverlay>}

'use client';

import { useEffect, useRef } from 'react';
import { Chart, sample } from '@/lib/material/chartjs';

/**
 * Gráfico de una función f(x) con su asíntota horizontal opcional.
 * `fn` debe ser una referencia estable (módulo), no una flecha creada en cada render.
 * `enabled` permite construirlo recién cuando el <details> se abre (lazy, como el original).
 */
export default function MathChart({
  fn, min, max,
  asymptote = null, asymptoteType = 'h',
  steps = 60, label = 'f(x)', enabled = true, className = 'h-44',
}) {
  const canvasRef = useRef(null);

  useEffect(() => {
    if (!enabled || !canvasRef.current) return;

    const { labels, ys } = sample(fn, min, max, steps);
    const datasets = [
      { label, data: ys, borderColor: '#2563eb', borderWidth: 2, pointRadius: 0, tension: 0.2 },
    ];
    if (asymptoteType === 'h' && asymptote !== null) {
      datasets.push({
        label: `Asíntota y = ${asymptote}`,
        data: ys.map(() => asymptote),
        borderColor: '#ef4444', borderWidth: 1.5, borderDash: [5, 5], pointRadius: 0,
      });
    }

    const chartInstance = new Chart(canvasRef.current, {
      type: 'line',
      data: { labels, datasets },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        plugins: { legend: { display: true, labels: { boxWidth: 12, font: { size: 10 } } } },
        scales: {
          x: { ticks: { maxTicksLimit: 6, font: { size: 9 } } },
          y: { ticks: { maxTicksLimit: 6, font: { size: 9 } } },
        },
      },
    });

    // Cleanup: StrictMode monta → desmonta → monta; sin destroy() Chart.js lanza "Canvas is already in use".
    return () => chartInstance.destroy();
  }, [enabled, fn, min, max, asymptote, asymptoteType, steps, label]);

  return (
    <div className={`relative mt-3 ${className}`}>
      <canvas ref={canvasRef} role="img" aria-label={`Gráfica de ${label}`} />
    </div>
  );
}

/** Renderiza las fórmulas de la página al montar (y tras navegar con el router, que no recarga scripts). */
export default function MathJaxTypeset() {
  useEffect(() => {
    const typeset = () => {
      const mj = window.MathJax;
      if (!mj?.startup?.promise) return false; // el CDN aún no terminó de cargar
      mj.startup.promise
        .then(() => {
          mj.typesetClear?.();          // olvida nodos de páginas anteriores
          return mj.typesetPromise();
        })
        .then(() => window.dispatchEvent(new Event('mathjax-typeset'))) // el aside de trigonometría reajusta su escala
        .catch((err) => console.error('MathJax:', err));
      return true;
    };

    if (!typeset()) window.addEventListener('mathjax-ready', typeset, { once: true });
    return () => window.removeEventListener('mathjax-ready', typeset);
  }, []);

  return null;
}

'use client';

import { createContext, useCallback, useContext, useEffect, useMemo, useState } from 'react';

const ProgressContext = createContext(null);

/**
 * Progreso de ejercicios resueltos, persistido en localStorage.
 * Mismas claves que la versión original (`math_exercise_ex1`…), así nadie pierde su avance.
 */
export function ProgressProvider({ total, prefix = 'math_exercise_', children }) {
  const [done, setDone] = useState({}); // { ex1: true, ... }

  // Lectura al montar: localStorage no existe en el servidor, por eso va en useEffect
  // (el primer render coincide con el HTML estático y no hay error de hidratación).
  useEffect(() => {
    const saved = {};
    try {
      for (let i = 1; i <= total; i++) {
        if (localStorage.getItem(`${prefix}ex${i}`) === 'true') saved[`ex${i}`] = true;
      }
    } catch {}
    setDone(saved);
  }, [total, prefix]);

  const toggle = useCallback((id) => {
    const next = !done[id];
    setDone((prev) => ({ ...prev, [id]: next }));
    try { localStorage.setItem(`${prefix}${id}`, String(next)); } catch {}
  }, [done, prefix]);

  const reset = useCallback(() => {
    if (!window.confirm('¿Deseas reiniciar todas las marcas de ejercicios resueltos?')) return;
    try {
      for (let i = 1; i <= total; i++) localStorage.removeItem(`${prefix}ex${i}`);
    } catch {}
    setDone({});
  }, [total, prefix]);

  const value = useMemo(() => ({
    total,
    count: Object.values(done).filter(Boolean).length,
    isDone: (id) => !!done[id],
    toggle,
    reset,
  }), [total, done, toggle, reset]);

  return <ProgressContext.Provider value={value}>{children}</ProgressContext.Provider>;
}

export function useProgress() {
  const ctx = useContext(ProgressContext);
  if (!ctx) throw new Error('useProgress debe usarse dentro de <ProgressProvider>');
  return ctx;
}

/** Contador "0 / 26 hechos" del header. */
export function HeaderProgress() {
  const { count, total } = useProgress();
  return (
    <div className="flex items-center gap-2 rounded-lg border border-slate-700 bg-slate-800 px-3 py-1">
      <span className="text-xs text-slate-300">Progreso:</span>
      <span className="font-mono text-xs font-bold text-emerald-400">{count} / {total} hechos</span>
    </div>
  );
}

'use client';

import { useEffect, useRef } from 'react';
import { Chart, sample } from '@/lib/material/chartjs';

/** f, f⁻¹ y el eje de simetría y = x en un mismo gráfico. */
export default function SymmetryChart({
  f, inverse, min, max, yMin, yMax, labels: names,
  step = 0.25, enabled = true, className = 'h-64',
}) {
  const canvasRef = useRef(null);

  useEffect(() => {
    if (!enabled || !canvasRef.current) return;

    const steps = Math.round((max - min) / step);
    const { xs, labels, ys: fData } = sample(f, min, max, steps);
    const { ys: invData } = sample(inverse, min, max, steps); // fuera del dominio → null

    const line = (label, data, borderColor, extra = {}) => ({
      label, data, borderColor, borderWidth: 2, pointRadius: 0, ...extra,
    });

    const chartInstance = new Chart(canvasRef.current, {
      type: 'line',
      data: {
        labels,
        datasets: [
          line(names.f, fData, '#2563eb'),
          line(names.inverse, invData, '#10b981'),
          line('Eje de Simetría y = x', xs, '#94a3b8', { borderDash: [4, 4], borderWidth: 1.5 }),
        ],
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        scales: { y: { min: yMin, max: yMax }, x: { ticks: { maxTicksLimit: 10 } } },
      },
    });

    return () => chartInstance.destroy();
  }, [enabled, f, inverse, min, max, yMin, yMax, names, step]);

  return (
    <div className={`relative w-full ${className}`}>
      <canvas ref={canvasRef} role="img" aria-label={`Gráfica de ${names.f} y su inversa`} />
    </div>
  );
}

/** Muestrea fn en [min, max]. Valores fuera del dominio (NaN/±Infinity) pasan a null = hueco en la curva. */
export function sample(fn, min, max, steps) {
  const xs = Array.from({ length: steps + 1 }, (_, i) => min + (i * (max - min)) / steps);
  return { xs, labels: xs.map((x) => x.toFixed(2)), ys: xs.map((x) => { const y = fn(x); return Number.isFinite(y) ? y : null; }) };
}

import { SITE_NAME, abs } from '@/lib/site';

/**
 * Metadata completa por página (title, canonical, OpenGraph, Twitter).
 * Next reemplaza `openGraph`/`twitter`/`alternates` de forma SUPERFICIAL entre layout y page,
 * así que cada page.js los define completos con este helper en vez de heredarlos a medias.
 */
export function buildMetadata({
  title,
  description,
  path = '/',
  ogTitle = title,
  ogDescription = description,
  twitterDescription = description,
  type = 'website',
  image = '/og-image.jpg',
}) {
  const url = abs(path);
  const imageUrl = abs(image);

  return {
    title: { absolute: title },
    description,
    alternates: { canonical: url },
    openGraph: {
      type,
      locale: 'es_AR',
      url,
      siteName: SITE_NAME,
      title: ogTitle,
      description: ogDescription,
      images: [
        {
          url: imageUrl,
          width: 1200,
          height: 630,
          alt: 'Clases particulares de Matemática en Palermo, Buenos Aires',
        },
      ],
    },
    twitter: {
      card: 'summary_large_image',
      title: ogTitle,
      description: twitterDescription,
      images: [imageUrl],
    },
  };
}

/** Ruta de un archivo de /public con basePath (para <img> y <a download>). */
export const asset = (path) => `${process.env.NEXT_PUBLIC_BASE_PATH ?? ''}${path}`;

/** URL absoluta (canonical, OG, JSON-LD). Metadata no antepone basePath, por eso se arma a mano. */
export const abs = (path = '/') => `${SITE_URL}${path.startsWith('/') ? path : `/${path}`}`;

/** Enlace de WhatsApp. Sin texto => enlace simple (como el botón flotante). */
export const whatsappUrl = (text) => {
  const base = `https://api.whatsapp.com/send/?phone=${PHONE}`;
  return text ? `${base}&text=${encodeURIComponent(text).replace(/!/g, '%21')}` : base;
};
